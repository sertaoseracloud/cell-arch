<objective>
Build the reusable `aws-eks` Terraform module that provisions a private EKS cluster, a managed node group (t3.medium × 2), the OIDC identity provider for IRSA, and the three required managed add-ons (vpc-cni, coredns, kube-proxy). Purpose: TERR-03 requires EKS with IRSA OIDC provider. RESEARCH.md Pattern 1 provides the verified IRSA pattern. The dynamic tls_certificate data source (Pitfall 1) prevents OIDC thumbprint drift. The node group dependency on add-ons (Pitfall 4) prevents CoreDNS degraded state.

Output: infra/modules/aws-eks/ with main.tf, variables.tf, outputs.tf — consumed by infra/live/aws/{dev,hom,prod}/.
</objective>

# ── IAM: EKS Cluster Role ────────────────────────────────────────────────────

data "aws_iam_policy_document" "eks_cluster_assume" {
  statement {
    effect  = "Allow"
    actions = ["sts:AssumeRole"]
    principals {
      type        = "Service"
      identifiers = ["eks.amazonaws.com"]
    }
  }

resource "aws_iam_role" "eks_cluster" {

# ── EKS Cluster ──────────────────────────────────────────────────────────────

resource "aws_eks_cluster" "main" {
  name     = "${var.project_name}-eks-${var.environment}"
  version  = var.kubernetes_version
  role_arn = aws_iam_role.eks_cluster.arn

  vpc_config {
    subnet_ids              = var.private_subnet_ids
    endpoint_private_access = true
    endpoint_public_access  = false
  }

  enabled_cluster_log_types = ["api", "audit", "authenticator"]

  tags = merge(var.tags, { Name = "${var.project_name}-eks-${var.environment}" })

  depends_on = [aws_iam_role_policy_attachment.eks_cluster_policy]
}

# ── Managed Node Group ────────────────────────────────────────────────────────

resource "aws_eks_node_group" "main" {
  cluster_name    = aws_eks_cluster.main.name
  node_group_name = "${var.project_name}-ng-${var.environment}"
  node_role_arn   = aws_iam_role.eks_node.arn
  subnet_ids      = var.private_subnet_ids

  scaling_config {
    desired_size = var.node_count
    min_size     = var.node_count
    max_size     = var.node_count + 2
  }

  instance_types = ["t3.medium"]

  tags = merge(var.tags, { Name = "${var.project_name}-ng-${var.environment}" })

  depends_on = [
    aws_iam_role_policy_attachment.eks_node_policy,
    aws_iam_role_policy_attachment.eks_cni_policy,
    aws_iam_role_policy_attachment.eks_ecr_policy,
  ]
}

# ── OIDC Provider for IRSA ────────────────────────────────────────────────────
# Dynamic thumbprint via tls_certificate data source prevents thumbprint drift
# (RESEARCH.md Pitfall 1)

data "tls_certificate" "eks" {
  url = aws_eks_cluster.main.identity[0].oidc[0].issuer
}

resource "aws_iam_openid_connect_provider" "eks" {
  url             = aws_eks_cluster.main.identity[0].oidc[0].issuer
  client_id_list  = ["sts.amazonaws.com"]
  thumbprint_list = [data.tls_certificate.eks.certificates[0].sha1_fingerprint]
}

# ── IRSA IAM Role ─────────────────────────────────────────────────────────────
# Trust policy uses StringEquals on both :sub and :aud — no wildcards
# (RESEARCH.md Pattern 1; security requirement: overly broad trust policy)

data "aws_iam_policy_document" "irsa_trust" {
  statement {
    effect  = "Allow"
    actions = ["sts:AssumeRoleWithWebIdentity"]
    principals {
      type        = "Federated"
      identifiers = [aws_iam_openid_connect_provider.eks.arn]
    }
    condition {
      test     = "StringEquals"
      variable = "${replace(aws_iam_openid_connect_provider.eks.url, "https://", "")}:sub"
      values   = ["system:serviceaccount:${var.namespace}:${var.service_account_name}"]
    }
    condition {
      test     = "StringEquals"
      variable = "${replace(aws_iam_openid_connect_provider.eks.url, "https://", "")}:aud"
      values   = ["sts.amazonaws.com"]
    }
  }
}

resource "aws_iam_role" "irsa" {
  name               = "${var.project_name}-irsa-${var.environment}"
  assume_role_policy = data.aws_iam_policy_document.irsa_trust.json
  tags               = var.tags
}

# This role is created here; policy attachment happens in live root after DynamoDB table exists.

<threat_model>
  <trust_boundaries>
    <boundary name="EKS API server access">endpoint_public_access = false: API server is only reachable from within the VPC. No external kubectl access without VPN/bastion.</boundary>
    <boundary name="IRSA token exchange">Kubernetes OIDC token → IAM STS AssumeRoleWithWebIdentity. Trust policy uses exact ServiceAccount subject — no wildcard.</boundary>
    <boundary name="EC2 node instance profiles">Node IAM role has minimum required policies: EKSWorkerNodePolicy, EKS_CNI_Policy, ECRReadOnly. No admin or broad IAM permissions.</boundary>
  </trust_boundaries>

No wildcards. Any pod not using the exact ServiceAccount cannot assume the IRSA role."/>
    <threat id="T-03-03-02" category="Spoofing" component="aws_eks_cluster.main — public API endpoint" disposition="mitigate" mitigation_plan="endpoint_public_access = false ensures the API server has no public endpoint.
