**infra/modules/azure-aks/main.tf**

```hcl
terraform {
  required_providers {
    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~> 4.0"
    }
  }
}

# ── AKS Cluster ───────────────────────────────────────────────────────────────
# oidc_issuer_enabled AND workload_identity_enabled MUST both be true
# (RESEARCH.md Pattern 2 — one without the other does not activate WI)

resource "azurerm_kubernetes_cluster" "main" {
  name                      = "${var.project_name}-aks-${var.environment}"
  location                  = var.location
  resource_group_name       = var.resource_group_name
  dns_prefix                = "${var.project_name}-${var.environment}"
  kubernetes_version        = var.kubernetes_version
  oidc_issuer_enabled       = true
  workload_identity_enabled = true

  default_node_pool {
    name           = "system"
    node_count     = var.node_count
    vm_size        = "Standard_D2s_v3"
    vnet_subnet_id = var.subnet_id
  }

  identity {
    type = "SystemAssigned"
  }

  network_profile {
    network_plugin = "azure"
    network_policy = "azure"
  }

  tags = merge(var.tags, { Name = "${var.project_name}-aks-${var.environment}" })
}

# ── User Assigned Managed Identity for sidecar Workload Identity ──────────────

resource "azurerm_user_assigned_identity" "sidecar" {
  name                = "${var.project_name}-wi-${var.environment}"
  resource_group_name = var.resource_group_name
  location            = var.location
  tags                = var.tags
}

# ── Federated Identity Credential ─────────────────────────────────────────────
# Subject is case-sensitive: MUST match exact Kubernetes namespace and SA name
# (RESEARCH.md Pitfall 2). Use lowercase for both.
# NOTE: After terraform apply, allow 10-60s for Entra ID propagation before
# testing pod authentication (Pitfall 5).

resource "azurerm_federated_identity_credential" "sidecar" {
  name                = "${var.project_name}-fic-${var.environment}"
  resource_group_name = var.resource_group_name
  parent_id           = azurerm_user_assigned_identity.sidecar.id
  audience            = ["api://AzureADTokenExchange"]
  issuer              = azurerm_kubernetes_cluster.main.oidc_issuer_url
  # Subject format: system:serviceaccount:<namespace>:<serviceaccount-name>
  # Both namespace and serviceaccount-name MUST be lowercase
  subject = "system:serviceaccount:${var.namespace}:${var.service_account_name}"
}
```
