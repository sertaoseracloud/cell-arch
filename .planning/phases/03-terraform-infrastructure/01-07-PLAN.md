# ── cert-manager ──────────────────────────────────────────────────────────────
# Chart: oci://quay.io/jetstack/charts/cert-manager v1.20.2
# crds.enabled = true (NOT installCRDs — that flag was deprecated in v1.15)
# (RESEARCH.md Pattern 8)

resource "helm_release" "cert_manager" {
  name             = "cert-manager"
  repository       = "oci://quay.io/jetstack/charts"
  chart            = "cert-manager"
  version          = "v1.20.2"
  namespace        = "cert-manager"
  create_namespace = true
  wait             = true
  timeout          = 600

  set {
    name  = "crds.enabled"
    value = "true"
  }

  # Node group must be running before Helm can deploy (Pitfall 3 + Pitfall 4)
  depends_on = [module.eks]
}

# ── Secrets Store CSI Driver ──────────────────────────────────────────────────
# Base driver chart — must be installed before cloud provider chart
# syncSecret.enabled = true (secrets-management.md: secretObjects for K8s Secret sync)
# (RESEARCH.md Pattern 9)

resource "helm_release" "secrets_store_csi" {
  name             = "secrets-store-csi-driver"
  repository       = "https://kubernetes-sigs.github.io/secrets-store-csi-driver/charts"
  chart            = "secrets-store-csi-driver"
  version          = "1.6.0"
  namespace        = "kube-system"
  wait             = true
  timeout          = 300

  set {
    name  = "syncSecret.enabled"
    value = "true"
  }

  depends_on = [helm_release.cert_manager]
}

# ── AWS Secrets Store CSI Driver Provider ────────────────────────────────────
# Floats to latest — no version pin (upstream recommendation for AWS provider)
# (RESEARCH.md Pattern 9)

resource "helm_release" "csi_provider_aws" {
  name       = "csi-secrets-store-provider-aws"
  repository = "https://aws.github.io/secrets-store-csi-driver-provider-aws"
  chart      = "secrets-store-csi-driver-provider-aws"
  namespace  = "kube-system"
  wait       = true
  timeout    = 300
