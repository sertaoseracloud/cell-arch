```hcl
locals {

  resource_group_name = var.resource_group_name

}

module "aks" {
  source               = "../../../modules/azure-aks"
  project_name         = var.project_name
  environment          = var.environment
  location             = var.location
  resource_group_name  = var.resource_group_name
  subnet_id            = module.vnet.aks_subnet_id
  node_count           = var.node_count
  kubernetes_version   = var.kubernetes_version
  namespace            = var.namespace
  service_account_name = var.service_account_name
  tags                 = local.tags
}

module "cosmosdb" {
  source              = "../../../modules/azure-cosmosdb"
  project_name        = var.project_name
  environment         = var.environment
  location            = var.location
  resource_group_name = var.resource_group_name
  private_subnet_id   = module.vnet.privatelink_subnet_id
  spoke_vnet_id       = module.vnet.spoke_vnet_id
  database_name       = var.database_name
  container_name      = var.container_name
  tags                = local.tags
}

data "azurerm_client_config" "current" {}

# CosmosDB RBAC role assignment for Workload Identity
# account_name uses module.cosmosdb.account_name directly (NOT a tautological ternary)
# Built-in role: Cosmos DB Built-in Data Contributor (ID ends in ...0002)
resource "azurerm_cosmosdb_sql_role_assignment" "sidecar" {
  resource_group_name = var.resource_group_name
  account_name        = module.cosmosdb.account_name
  role_definition_id  = "/subscriptions/${data.azurerm_client_config.current.subscription_id}/resourceGroups/${var.resource_group_name}/providers/Microsoft.DocumentDB/databaseAccounts/${module.cosmosdb.account_name}/sqlRoleDefinitions/00000000-0000-0000-0000-000000000002"
  principal_id        = module.aks.workload_identity_object_id
  scope               = module.cosmosdb.resource_id
}
```

<threat_model>
  <trust_boundaries>
    <boundary name="Provider auth -- kubelogin exec block">Kubernetes and Helm providers use kubelogin exec-based token retrieval. Token is ephemeral -- not stored. No static credentials in providers.tf.</boundary>
    <boundary name="Azure Blob state lock">Blob Lease locking is automatic on azurerm backend -- prevents concurrent terraform apply from corrupting state. No extra configuration required.</boundary>
    <boundary name="CosmosDB RBAC scope">Role assignment scoped to CosmosDB account resource_id. Built-in Data Contributor role only grants data plane access -- not control plane (no ability to delete or modify the account).</boundary>
  </trust_boundaries>
  <threats>
    <threat id="T-03-09-01" category="Information Disclosure" component="Azure Blob state backend -- sensitive state outputs" disposition="mitigate" mitigation_plan="azurerm backend: container_access_type = private (set in bootstrap), min_tls_version = TLS1_2. Blob Lease provides concurrent-write protection. Azure Blob encryption at rest is always on."/>
    <threat id="T-03-09-02" category="Elevation of Privilege" component="providers.tf -- static Azure credentials" disposition="mitigate" mitigation_plan="No client_id, client_secret, or tenant_id hardcoded in providers.tf. Azure provider uses ambient auth (Workload Identity in CI, az login locally). Kubelogin exec block uses ephemeral Azure CLI tokens."/>
    <threat id="T-03-09-03" category="Elevation of Privilege" component="azurerm_cosmosdb_sql_role_assignment -- over-broad scope" disposition="mitigate" mitigation_plan="Role assignment uses Cosmos DB Built-in Data Contributor (ID ending in 0002) which grants data plane access only. Scope is the specific CosmosDB account resource_id. Principal is the AKS Workload Identity for the sidecar service account only."/>
    <threat id="T-03-09-04" category="Spoofing" component="Federated credential subject case sensitivity (Pitfall 2)" disposition="mitigate" mitigation_plan="namespace and service_account_name variables use lowercase defaults ('default', 'sidecar'). Executor MUST ensure Kubernetes ServiceAccount names match exactly. The federated credential in azure-aks module uses system:serviceaccount:{var.namespace}:{var.service_account_name} format which must exactly match the K8s object names (case-sensitive per Pitfall 2)."/>
